% Level-1 headings (numbered chapters and numberless front/back-matter chapters)
% are rendered in a single consistent style: uppercase and the same enlarged
% font size, so "CHAPTER 1 INTRODUCTION", "ABSTRACT", "APPROVAL PAGE", and
% "REFERENCES" all match. The \MakeUppercase before-code receives the title as
% its argument (a documented titlesec behaviour) and uppercases it.
\titleformat{\chapter}[block]
  {\normalfont\bfseries\fontsize{16pt}{19pt}\selectfont\RaggedRight}
  {\MakeUppercase{Chapter \thechapter}}
  {0.6em}
  {\MakeUppercase}
\titleformat{name=\chapter,numberless}[block]
  {\normalfont\bfseries\fontsize{16pt}{19pt}\selectfont\RaggedRight}
  {}
  {0pt}
  {\MakeUppercase}
\titlespacing*{\chapter}{0pt}{0pt}{1.5\baselineskip}

\titleformat{\section}
  {\normalfont\bfseries\normalsize\RaggedRight}
  {\thesection}
  {0.75em}
  {}
\titleformat{\subsection}
  {\normalfont\bfseries\normalsize\RaggedRight}
  {\thesubsection}
  {0.75em}
  {}
\titleformat{\subsubsection}
  {\normalfont\bfseries\normalsize\RaggedRight}
  {\thesubsubsection}
  {0.75em}
  {}

% Keep body headings in title case, but write numbered chapter titles to the
% Table of Contents in uppercase. Child section entries such as 1.1, 2.1, and
% 3.x remain exactly as written.
\makeatletter
\let\thesis@chapter\chapter
\RenewDocumentCommand{\chapter}{s o m}{%
  \IfBooleanTF{#1}{%
    \thesis@chapter*{#3}%
  }{%
    \IfNoValueTF{#2}{%
      \thesis@chapter[\MakeUppercase{#3}]{#3}%
    }{%
      \thesis@chapter[\MakeUppercase{#2}]{#3}%
    }%
  }%
}
\makeatother
